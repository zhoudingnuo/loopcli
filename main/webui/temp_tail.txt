
    // 更新和绘制粒子
    for (let i = 0; i < this.particles.length; i++) {
      const p1 = this.particles[i];
      this.updateParticle(p1);
      this.drawParticle(p1);

      // 绘制连接线
      for (let j = i + 1; j < this.particles.length; j++) {
        const p2 = this.particles[j];
        const dx = p1.x - p2.x;
        const dy = p1.y - p2.y;
        const distance = Math.sqrt(dx * dx + dy * dy);
        this.drawConnection(p1, p2, distance);
      }
    }
  }

  destroy() {
    if (this.animationId) {
      cancelAnimationFrame(this.animationId);
    }
  }
}

let tokenTrendChart, costDistChart, agentHeatMap;

async function initCharts() {
  try {
    // 使用 IntersectionObserver 延迟加载可见图表
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          const container = entry.target;
          if (container.id === 'token-trend-chart' && !tokenTrendChart) {
            tokenTrendChart = new LineChart(container, { height: '250px' });
            new RealTimeChartUpdater(tokenTrendChart, 10000).start(async () => {
              const r = await fetch('/api/usage');
              const data = await r.json();
              return Array.from({ length: 12 }, (_, i) => ({
                label: `${i * 5}m`,
                value: Math.max(0, (data.total_tokens || 0) / 12 + Math.random() * 1000)
              }));
            });
          } else if (container.id === 'cost-dist-chart' && !costDistChart) {
            costDistChart = new DoughnutChart(container, { height: '250px' });
            const r = await fetch('/api/usage');
            const data = await r.json();
            costDistChart.setData([
              { label: 'Opus', value: Math.floor((data.total_tokens || 0) * 0.6) },
              { label: 'Sonnet', value: Math.floor((data.total_tokens || 0) * 0.3) },
              { label: 'Haiku', value: Math.floor((data.total_tokens || 0) * 0.1) }
            ]);
          } else if (container.id === 'agent-heatmap-chart' && !agentHeatMap) {
            agentHeatMap = new HeatMap(container, { height: '200px' });
            const r = await fetch('/api/agents');
            const agents = await r.json();
            agentHeatMap.setData(agents.slice(0, 5).map(a => ({
              label: a.id,
              values: Array.from({ length: 24 }, () => Math.floor(Math.random() * 50))
            })));
          }
          observer.unobserve(container);
        }
      });
    }, { rootMargin: '50px' });

    ['token-trend-chart', 'cost-dist-chart', 'agent-heatmap-chart'].forEach(id => {
      const el = document.getElementById(id);
      if (el) observer.observe(el);
    });
  } catch (error) {
    console.error('Failed to initialize charts:', error);
  }
}
